.device atmega328p
.equ DDRB = 4
.equ PORTB = 5
main:
    ldi r16,$30 ; putting pin 12 address
    out DDRB,r16 ; making it out put

repeat: ;Calling NOTE to make music
    rcall NOTE_C
    rcall NOTE_C
    rcall NOTE_G
    rcall NOTE_G
    rcall NOTE_A
    rcall NOTE_A
    rcall NOTE_G
    rcall NOTE_F
    rcall NOTE_F
    rcall NOTE_E
    rcall NOTE_E
    rcall NOTE_D
    rcall NOTE_D
    rcall NOTE_C
rjmp repeat ; start over


; Notes

NOTE_C:
    ldi r20, 130
    NOTE_C_LOOP:
        ;NOTE C Part
        ldi r16,$30 ; putting 5v inside of it (using 1 for the pin 12)
        out PORTB,r16 ; making it happen
        rcall NOTE_C_DELAY ; delay for some time
        ldi r16,$00 ; putting 0v inside of it (using 0 for the pin 12)
        out PORTB, r16 ; making it happen
        rcall NOTE_C_DELAY ; delay it for some time
        dec r20
        brne NOTE_C_LOOP
    ret

; This is a nested for (Supposedly)
NOTE_C_DELAY: ; name (label) of the loop or function
    ldi r17, 1 ; first FOR counter
NOTE_C_1:
    ldi r18, 40 ; second FOR counter
NOTE_C_2:
    ldi r19, 250 ; third FOR counter
NOTE_C_3:
    dec r19 ; it decreases until 0 (inner loop of our nested loop)
    brne NOTE_C_3 ; it checks whether r18 is 0 or not and continue if it was 

    dec r18 ; it decreases until 0 (Middle inner loop)
    brne NOTE_C_2; if it was not 0 then it will again set the value into the r18

    dec r17 ; it decreases until 0 (outer loop)
    brne NOTE_C_1

    ret ; if the r17 is zero then the loop is over and it goes to the rcall and continues 


NOTE_D_DELAY: ; same thing as NOTE1 but with different counter and delay
